import os, json, uuid, re, hmac, hashlib, time, logging
from datetime import datetime
from collections import defaultdict
import pandas as pd
from sqlalchemy import create_engine, Column, String, Float, Integer, ForeignKey, DateTime
from sqlalchemy.orm import declarative_base, sessionmaker

# ------------------------------------------------------------------------------
# CONFIGURATION
# ------------------------------------------------------------------------------
DATABASE_URL = "sqlite:///shieldai_v34_universal.db"  # SQLite for demo; swap to PostgreSQL in production
SECRET_SALT = b"enterprise-salt-key"

engine_db = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(bind=engine_db, autocommit=False, autoflush=False)
Base = declarative_base()

logging.basicConfig(level=logging.INFO)

# ------------------------------------------------------------------------------
# DATA SCHEMAS
# ------------------------------------------------------------------------------
class ClientTenant(Base):
    __tablename__ = "clients"
    api_key_hash = Column(String(64), primary_key=True)
    company_name = Column(String(100), nullable=False)
    industry_sector = Column(String(50), nullable=False)
    tier = Column(String(50), nullable=False)
    rate_per_scan = Column(Float, nullable=False)

class TransactionLedger(Base):
    __tablename__ = "transactions"
    tx_id = Column(String(36), primary_key=True)
    api_key_hash = Column(String(64), ForeignKey("clients.api_key_hash"))
    timestamp = Column(DateTime, default=datetime.utcnow)
    status = Column(String(20))
    threat_level = Column(String(20))
    cost_usd = Column(Float)

class SecurityIncident(Base):
    __tablename__ = "incidents"
    incident_id = Column(Integer, primary_key=True, autoincrement=True)
    tx_id = Column(String(36))
    api_key_hash = Column(String(64), ForeignKey("clients.api_key_hash"))
    timestamp = Column(DateTime, default=datetime.utcnow)
    threat_type = Column(String(50))
    severity = Column(String(20))
    attacker_ip = Column(String(45))

Base.metadata.create_all(engine_db)

# ------------------------------------------------------------------------------
# UTILITIES
# ------------------------------------------------------------------------------
def hash_api_key(api_key: str) -> str:
    return hmac.new(SECRET_SALT, api_key.strip().encode(), hashlib.sha256).hexdigest()

class RateLimiter:
    def __init__(self, max_requests: int = 100, window_seconds: int = 60):
        self.max_requests = max_requests
        self.window = window_seconds
        self.requests = defaultdict(list)

    def is_allowed(self, client_id: str) -> bool:
        now = time.time()
        self.requests[client_id] = [t for t in self.requests[client_id] if now - t < self.window]
        if len(self.requests[client_id]) < self.max_requests:
            self.requests[client_id].append(now)
            return True
        return False

rate_limiter = RateLimiter()

# ENGINE
# ------------------------------------------------------------------------------
class OmniShieldEngine:
    def __init__(self):
        self.regex_signatures = [
            r"(system\s*override)",
            r"(ignore|bypass)\s+(rules|filters|instructions)",
            r"(reveal|expose)\s+(password|key|configuration)"
        ]

    def process_transaction(self, api_key: str, user_message: str, guard_mode: str):
        session = SessionLocal()
        tx_id = str(uuid.uuid4())
        attacker_ip = "127.0.0.1"
        timestamp_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

        try:
            hashed_key = hash_api_key(api_key)
            client = session.query(ClientTenant).filter_by(api_key_hash=hashed_key).first()
            if not client:
                return {"status": "ACCESS_DENIED", "message": "Invalid API Key"}

            if not rate_limiter.is_allowed(hashed_key):
                return {"status": "RATE_LIMITED", "message": "429 Too Many Requests"}

            clean_text = user_message.strip()
            if not clean_text:
                return {"status": "IDLE", "message": "Empty payload"}

            regex_threat = any(re.search(p, clean_text, re.IGNORECASE) for p in self.regex_signatures)

            if regex_threat or guard_mode == "Force Decoy":
                incident = SecurityIncident(tx_id=tx_id, api_key_hash=hashed_key,
                                            threat_type="REGEX_MATCH", severity="HIGH",
                                            attacker_ip=attacker_ip)
                session.add(incident)
                session.commit()
                return {"status": "BLOCKED", "message": "Threat blocked"}

            # Safe route -> Answer with LLM
            answer = self._answer_with_llm(clean_text, client.company_name, client.industry_sector)
            tx = TransactionLedger(tx_id=tx_id, api_key_hash=hashed_key,
                                   status="APPROVED", threat_level="NONE",
                                   cost_usd=client.rate_per_scan)
            session.add(tx)
            session.commit()
            return {"status": "APPROVED", "message": answer}

        except Exception as e:
            logging.error("Error: %s", str(e))
            return {"status": "ERROR", "message": str(e)}
        finally:
            session.close()

    def _answer_with_llm(self, text, company, sector):
        try:
            import openai
            openai.api_key = os.getenv("OPENAI_API_KEY")
            if not openai.api_key:
                return f"[Safe payload for {company}] No LLM key set. Please configure OPENAI_API_KEY."
            response = openai.ChatCompletion.create(
                model="gpt-4o-mini",
                messages=[
                    {"role": "system", "content": f"You are a helpful assistant for {company} in the {sector} industry."},
                    {"role": "user", "content": text}
                ]
            )
            return response.choices[0].message["content"]
        except Exception as e:
            return f"[Safe payload for {company}] LLM error: {str(e)}"

def seed_demo_clients():
    session = SessionLocal()
    demo_clients = [
        ("demo_bank", "Habib Bank Limited", "Banking", "Enterprise", 0.005),
        ("demo_retail", "Daraz E-Commerce", "Retail", "Professional", 0.008),
        ("demo_telco", "Telenor Pakistan", "Telecom", "Enterprise", 0.005),
        ("demo_health", "Shaukat Khanum Hospital", "Healthcare", "Enterprise", 0.004)
    ]
    for key, name, sector, tier, rate in demo_clients:
        hashed = hash_api_key(key)
        if not session.query(ClientTenant).filter_by(api_key_hash=hashed).first():
            client = ClientTenant(api_key_hash=hashed, company_name=name,
                                  industry_sector=sector, tier=tier, rate_per_scan=rate)
            session.add(client)
    session.commit()
    session.close()
